function regenerateMembershipCard() {

  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getActiveSheet();

  const row = sheet.getActiveCell().getRow();

  if (row <= 1) {
    SpreadsheetApp.getUi().alert(
      "Please select a member row."
    );
    return;
  }

  generateMembershipCard(row);

}
function generateQRBlob(qrText) {

  const qrURL =
    "https://api.qrserver.com/v1/create-qr-code/?size=300x300&data=" +
    encodeURIComponent(qrText);

  const response = UrlFetchApp.fetch(qrURL);

  return response.getBlob();
}
// placeholder
function replacePlaceholderWithImage(slide, placeholder, imageBlob) {

  const pageElements = slide.getPageElements();

  for (let i = 0; i < pageElements.length; i++) {

    const element = pageElements[i];

    if (element.getPageElementType() ==
        SlidesApp.PageElementType.SHAPE) {

      const shape = element.asShape();

      const text = shape.getText().asString().trim();

      if (text == placeholder) {

        const left = element.getLeft();
        const top = element.getTop();
        const width = element.getWidth();
        const height = element.getHeight();

        element.remove();

        const image = slide.insertImage(imageBlob);

        image.setLeft(left);
        image.setTop(top);
        const originalWidth = image.getWidth();
const originalHeight = image.getHeight();

const scale = Math.max(
  width / originalWidth,
  height / originalHeight
);

image.setWidth(originalWidth * scale);
image.setHeight(originalHeight * scale);

image.setLeft(left + (width - image.getWidth()) / 2);
image.setTop(top + (height - image.getHeight()) / 2);

        break;

      }
    }
  }
}


function getDriveFileId(url) {

  const match = url.match(/[-\w]{25,}/);

  if (match) {
    return match[0];
  }

  return null;

}

/**
 * Limits text to a maximum number of characters.
 * It never cuts a word in half.
 */
function shortenText(text, maxLength) {

  // If already short enough, return as it is
  if (text.length <= maxLength) {
    return text;
  }

  // Cut at the maximum length
  let shortened = text.substring(0, maxLength);

  // Find the last space
  const lastSpace = shortened.lastIndexOf(" ");

  // If a space exists, remove the incomplete last word
  if (lastSpace > 0) {
    shortened = shortened.substring(0, lastSpace);
  }

  return shortened;
}


/**
 * Returns the next available PDF name.
 *
 * Example:
 * First PDF:
 * CBS-NMU-2026-001.pdf
 *
 * Second:
 * CBS-NMU-2026-001_v2.pdf
 *
 * Third:
 * CBS-NMU-2026-001_v3.pdf
 */
function getNextPdfName(folder, memberID) {

  let highestVersion = 0;

  const files = folder.getFiles();

  while (files.hasNext()) {

    const file = files.next();

    const name = file.getName();

    // Ignore files that don't belong to this member
    if (!name.startsWith(memberID)) {
      continue;
    }

    // Original PDF (no version)
    if (name === memberID + ".pdf") {

      highestVersion = Math.max(highestVersion, 1);
      continue;

    }

    // Match filenames like:
    // CBS-NMU-2026-001_v2.pdf
    // CBS-NMU-2026-001_v15.pdf

    const match =
      name.match(/_v(\d+)\.pdf$/);

    if (match) {

      const version =
        parseInt(match[1], 10);

      highestVersion =
        Math.max(highestVersion, version);

    }

  }

  // First generation
  if (highestVersion === 0) {

    return memberID + ".pdf";

  }

  // Next version
  return memberID + "_v" +
         (highestVersion + 1) +
         ".pdf";

}
